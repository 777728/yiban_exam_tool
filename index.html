# -*- coding: utf-8 -*-
"""
校纪校规考试成绩汇总转换工具
================================
把易班导出的 ExamCompletionReport 类 CSV（GBK 编码）按
"云南大学2026级第一次校纪校规考试成绩汇总"模板结构，
转换为 Excel 汇总文件夹。

用法（命令行）:
    python convert_exam.py --csv ExamCompletionReport20261019234636.csv \
        --type grad --folder "云南大学2026级第一次研究生博士生校纪校规考试成绩汇总"

    --type:  undergrad | grad
             （本科生 / 研究生博士生；研究生、博士均可归入 grad）
    --folder: 输出文件夹名称（同时也是表头生成的依据）
    --out:    输出根目录（默认当前目录）
    --pass-line: 通过分数线，默认 90

也可作为模块调用:
    from convert_exam import process_exam
    result = process_exam(csv_path, exam_type="grad",
                          folder_name="云南大学2026级第一次研究生博士生校纪校规考试成绩汇总")
"""

import os
import re
import csv
import argparse

from openpyxl import Workbook
from openpyxl.styles import Font, Alignment, PatternFill, Border, Side
from openpyxl.utils import get_column_letter

# ----------------------------- 常量 -----------------------------
PASS_LINE = 90          # 通过分数线（依据模板：通过最低 90，未通过最高 88）
EXAM_WORD = "校规校纪考试"   # 考试名称用词

# 表头四列（固定顺序：姓名、学号、学院、成绩）
COLS = ["姓名", "学号", "学院", "成绩"]

# 样式
TITLE_FONT = Font(name="微软雅黑", size=14, bold=True)
HEAD_FONT = Font(name="微软雅黑", size=11, bold=True, color="FFFFFF")
HEAD_FILL = PatternFill("solid", fgColor="4472C4")
TITLE_FILL = PatternFill("solid", fgColor="D9E1F2")
CENTER = Alignment(horizontal="center", vertical="center", wrap_text=True)
LEFT = Alignment(horizontal="left", vertical="center")
THIN = Side(style="thin", color="BFBFBF")
BORDER = Border(left=THIN, right=THIN, top=THIN, bottom=THIN)


# ----------------------------- 工具函数 -----------------------------
def detect_encoding(path):
    """尝试常见编码读取，优先正确解码中文 GBK。"""
    for enc in ("gbk", "utf-8-sig", "utf-8", "latin-1"):
        try:
            with open(path, encoding=enc) as f:
                f.read(4096)
            return enc
        except UnicodeDecodeError:
            continue
    return "utf-8"


def clean_sid(raw):
    """清洗学号：去掉反引号、空格等。"""
    return (raw or "").replace("`", "").replace("'", "").strip()


def parse_score(raw_score):
    """从 '100 / 100.0%' 或 '100.0' 中解析出分数（数值）。"""
    if raw_score is None:
        return 0.0
    m = re.search(r"(\d+(?:\.\d+)?)\s*/", str(raw_score))
    if m:
        return float(m.group(1))
    try:
        return float(str(raw_score).strip())
    except ValueError:
        return 0.0


def infer_sid(sid):
    """
    解析学号，返回 (type, year)。
      type: 'undergrad' | 'master' | 'phd' | None(无法识别)
      year: 4 位年份字符串，如 '2026'；无法识别为 None
    规则（关键：本科与博士都以 '2' 开头，用 '20' 前缀优先区分）:
      本科: 11位, 以 '20' 开头 -> 前4位为年份
      硕士: 11位, 以 '1'  开头 -> '1'+4位年份+6位尾数
      博士: 11位, 以 '2'  开头 -> '2'+4位年份+6位尾数
    """
    if not sid:
        return None, None
    if len(sid) == 11 and sid.startswith("20"):
        return "undergrad", sid[0:4]
    if len(sid) == 11 and sid.startswith("1"):
        return "master", sid[1:5]
    if len(sid) == 11 and sid.startswith("2"):
        return "phd", sid[1:5]
    return None, None


def extract_target_year(folder_name):
    """从文件夹名称中提取目标年级年份，如 '2026级' -> 2026。"""
    m = re.search(r"(\d{4})级", folder_name)
    if m:
        return int(m.group(1))
    return 2026


def norm_type_arg(t):
    """归一化类型参数。"""
    t = (t or "").strip()
    if t in ("undergrad", "本科生", "本科"):
        return "undergrad"
    if t in ("grad", "研究生博士生", "研究生", "博士", "研究生/博士生"):
        return "grad"
    return None


# ----------------------------- 核心处理 -----------------------------
def process_exam(csv_path, exam_type, folder_name, out_root=None, pass_line=PASS_LINE):
    """
    主处理函数。

    返回 dict:
        status: 'ok' | 'upload_error'
        msg: 说明
        out_dir: 输出文件夹绝对路径
        stats: 统计摘要
    """
    exam_type = norm_type_arg(exam_type)
    if exam_type is None:
        return {"status": "upload_error",
                "msg": "类型参数错误，应为 'undergrad'(本科生) 或 'grad'(研究生博士生)。"}
    if not folder_name:
        return {"status": "upload_error", "msg": "缺少输出文件夹名称。"}
    if not os.path.exists(csv_path):
        return {"status": "upload_error", "msg": f"找不到文件：{csv_path}"}

    target_year = extract_target_year(folder_name)
    declared_is_grad = (exam_type == "grad")

    # ---- 1. 读取 CSV ----
    enc = detect_encoding(csv_path)
    with open(csv_path, encoding=enc) as f:
        reader = csv.reader(f)
        rows = [r for r in reader if any(c.strip() for c in r)]
    if not rows:
        return {"status": "upload_error", "msg": "文件为空。"}

    header = rows[0]
    # 列名映射（容错：包含关键字即可）
    def find_col(*keys):
        for i, h in enumerate(header):
            for k in keys:
                if k in (h or ""):
                    return i
        return None

    i_name = find_col("姓名")
    i_sid = find_col("学号")
    i_college = find_col("学院")
    i_status = find_col("状态")
    i_score = find_col("成绩")
    if i_score is None:
        # 退而求其次：含 '百分比' 或 '得分率' 的列
        i_score = find_col("百分比", "得分率")
    if None in (i_name, i_sid, i_college, i_score):
        miss = [n for n, idx in (("姓名", i_name), ("学号", i_sid),
                                 ("学院", i_college), ("成绩", i_score)) if idx is None]
        return {"status": "upload_error",
                "msg": f"文件缺少必要列：{', '.join(miss)}。请上传易班标准导出文件。"}

    # ---- 2. 逐行分类 ----
    records = []          # 全部有效记录（含分类信息）
    type_counter = {"undergrad": 0, "master": 0, "phd": 0, "none": 0}
    for r in rows[1:]:
        name = r[i_name].strip() if i_name < len(r) else ""
        sid_raw = r[i_sid] if i_sid < len(r) else ""
        college = (r[i_college].strip() if i_college < len(r) else "") or "未知学院"
        status = (r[i_status].strip() if i_status is not None and i_status < len(r) else "")
        # [对比验证] "未开始/进行中"人员成绩解析为 0，仍按未通过计入对应表，
        # 与模板主流一致（模板全量主表纳入了 43/54 个未开始人员）。
        # 经核对，模板与源文件批次不一致会"漏"掉个别未开始/已完成人员，
        # 属模板侧数据缺失，程序按源文件完整计入，属正确行为，不在此排除。
        pts = parse_score(r[i_score] if i_score < len(r) else "")
        sid = clean_sid(sid_raw)
        itype, year = infer_sid(sid)

        rec = {
            "name": name, "sid": sid, "sid_raw": sid_raw,
            "college": college, "status": status, "pts": pts,
            "itype": itype, "year": year,
            "is_grad": itype in ("master", "phd"),
            "category": None, "reason": "",
        }
        type_counter[itype if itype else "none"] += 1

        if itype is None:
            rec["category"] = "format_error"
            rec["reason"] = "学号格式异常"
        # [对比验证] 以下分类规则已与标准模板"问题表"(39人)逐条核对一致：
        #   硕/博 或 F… 格式异常 -> cuo；本科且非目标年级 -> retake；本科且目标年级 -> valid。
        # 模板曾把 6 名博士/F 人员同时误放入"2026级本科生"valid 表（重复列示），
        # 此处只在 cuo 表，分类更规范、无重复，故保持原逻辑不动。
        elif rec["is_grad"] != declared_is_grad:
            rec["category"] = "cuo"
            rec["reason"] = ("研究生/博士生错考本科生题目"
                             if rec["is_grad"] else "本科生错考研究生/博士生题目")
        elif year == str(target_year):
            rec["category"] = "valid"
        else:
            rec["category"] = "retake"
        records.append(rec)

    # ---- 3. 文件级校验（类型不符 -> 上传错误）----
    grad_n = type_counter["master"] + type_counter["phd"]
    under_n = type_counter["undergrad"]
    if grad_n == 0 and under_n == 0:
        return {"status": "upload_error",
                "msg": "文件中没有可识别的学号，请检查上传文件是否为易班标准导出文件。"}
    # 以"主导类型"判断文件实际属于哪类考试
    dominant_is_grad = grad_n > under_n
    if declared_is_grad != dominant_is_grad:
        actual = "研究生/博士生" if dominant_is_grad else "本科生"
        chosen = "研究生博士生" if declared_is_grad else "本科生"
        return {"status": "upload_error",
                "msg": f"文件类型与所选[{chosen}]不符：文件实际为[{actual}]考试文件"
                       f"（识别到研究生/博士生 {grad_n} 人、本科生 {under_n} 人），"
                       f"请重新选择类型或文件。"}
    if len(records) == 0:
        return {"status": "upload_error", "msg": "文件中没有任何有效数据行。"}

    # ---- 4. 分组 ----
    valid = [r for r in records if r["category"] == "valid"]
    retake = [r for r in records if r["category"] == "retake"]
    problem = [r for r in records if r["category"] in ("cuo", "format_error")]

    def sort_key(r):
        return -r["pts"]

    valid.sort(key=sort_key)
    retake.sort(key=sort_key)
    problem.sort(key=sort_key)

    # 分学院（仅本年有效）
    by_college = {}
    for r in valid:
        by_college.setdefault(r["college"], []).append(r)

    # ---- 5. 写文件 ----
    if out_root is None:
        out_root = os.getcwd()
    out_dir = os.path.join(out_root, folder_name)
    att1 = os.path.join(out_dir, f"附件1-{folder_name}-各学院通过率统计表")
    att2 = os.path.join(out_dir, f"附件2-{folder_name}-情况统计表")
    os.makedirs(att1, exist_ok=True)
    att2_college = os.path.join(att2, f"01--{folder_name}-分学院")
    os.makedirs(att2_college, exist_ok=True)

    # 5a. 分学院：每个学院一个 xlsx，含 通过/未通过 两个 sheet
    for college, recs in by_college.items():
        passed = [r for r in recs if r["pts"] >= pass_line]
        failed = [r for r in recs if r["pts"] < pass_line]
        wb = _new_wb()
        _fill_list_sheet(wb, "通过", f"{folder_name}通过名单", passed, has_problem=False)
        _fill_list_sheet(wb, "未通过", f"{folder_name}未通过名单", failed, has_problem=False)
        safe = re.sub(r"[\\/:*?\"<>|]", "_", college)
        wb.save(os.path.join(att2_college, f"{folder_name}-{safe}-{EXAM_WORD}名单.xlsx"))

    # 5b. 往届补考名单（通过/未通过）
    wb = _new_wb()
    _fill_list_sheet(wb, "通过", f"{folder_name}往届补考通过名单",
                     [r for r in retake if r["pts"] >= pass_line], has_problem=False)
    _fill_list_sheet(wb, "未通过", f"{folder_name}往届补考未通过名单",
                     [r for r in retake if r["pts"] < pass_line], has_problem=False)
    wb.save(os.path.join(att2, f"02-{folder_name}-往届补考名单.xlsx"))

    # 5c. 错考/问题名单（含 问题 列）
    wb = _new_wb()
    _fill_list_sheet(wb, "问题", f"{folder_name}问题名单", problem, has_problem=True)
    wb.save(os.path.join(att2, f"03-{folder_name}-错考名单.xlsx"))

    # 5d. 通过率统计表
    _write_pass_rate(wb_path=os.path.join(att1, f"01-{folder_name}-各学院通过率统计表.xlsx"),
                     records=records, folder_name=folder_name,
                     target_year=target_year, pass_line=pass_line)

    # ---- 6. 统计摘要 ----
    stats = _build_stats(records, folder_name, target_year, pass_line)
    return {
        "status": "ok",
        "msg": "处理完成",
        "out_dir": out_dir,
        "stats": stats,
    }


# ----------------------------- 写表辅助 -----------------------------
def _new_wb():
    """新建 workbook 并去掉默认的空 Sheet。"""
    wb = Workbook()
    default = wb.sheetnames[0]
    wb.remove(wb[default])
    return wb


def _fill_list_sheet(wb, sheet_name, title, recs, has_problem):
    """在 workbook 中写入一个 sheet（若已存在则复用，否则新建）。"""
    if sheet_name in wb.sheetnames:
        ws = wb[sheet_name]
    else:
        ws = wb.create_sheet(sheet_name)
    ws.delete_rows(1, ws.max_row)  # 清空（仅新建的空白 sheet 无影响）

    ncol = 5 if has_problem else 4
    # 标题行
    ws.merge_cells(start_row=1, start_column=1, end_row=1, end_column=ncol)
    c = ws.cell(row=1, column=1, value=title)
    c.font = TITLE_FONT
    c.alignment = CENTER
    c.fill = TITLE_FILL
    # 表头
    headers = COLS + (["问题"] if has_problem else [])
    for j, h in enumerate(headers, 1):
        cell = ws.cell(row=2, column=j, value=h)
        cell.font = HEAD_FONT
        cell.fill = HEAD_FILL
        cell.alignment = CENTER
        cell.border = BORDER
    # 数据
    for i, r in enumerate(recs, start=3):
        sid_val = r["sid"]
        try:
            sid_val = int(sid_val)
        except (ValueError, TypeError):
            sid_val = r["sid"] or ""
        row_vals = [r["name"], sid_val, r["college"], int(r["pts"])]
        if has_problem:
            row_vals.append(r["reason"])
        for j, v in enumerate(row_vals, 1):
            cell = ws.cell(row=i, column=j, value=v)
            cell.alignment = CENTER if j != 1 else LEFT
            cell.border = BORDER
    # 列宽
    widths = [12, 16, 26, 8, 24] if has_problem else [12, 16, 26, 8]
    for j, w in enumerate(widths, 1):
        ws.column_dimensions[get_column_letter(j)].width = w
    ws.freeze_panes = "A3"


def _write_pass_rate(wb_path, records, folder_name, target_year, pass_line):
    wb = Workbook()
    ws = wb.active
    ws.title = "Sheet1"

    # 两个群体：本科生 / 研究生博士生
    blocks = [
        ("本科生", "undergrad"),
        ("研究生", "grad"),   # 研究生博士生群体（硕+博）
                              # [对比验证] 当本次文件为本科生考试时，研究生全量不在此文件，
                              # pool_for("grad") 仅含被错考的硕/博（已归 cuo，被下方 continue 跳过），
                              # 故研究生块为空，与模板"双块"结构对齐（模板研究生块来自另一研究生CSV）。
    ]

    def pool_for(pop):
        """参加考试池：该群体且非错考/非格式异常（即 valid+retake）。"""
        out = []
        for r in records:
            if r["category"] in ("cuo", "format_error"):
                continue
            if pop == "undergrad" and r["itype"] == "undergrad":
                out.append(r)
            elif pop == "grad" and r["itype"] in ("master", "phd"):
                out.append(r)
        return out

    def stats_of(pool):
        total = len(pool)
        p = sum(1 for r in pool if r["pts"] >= pass_line)
        f = total - p
        rate = (p / total) if total else 0
        return [total, p, f, rate]

    row = 1
    # 总标题
    ws.merge_cells(start_row=row, start_column=1, end_row=row, end_column=8)
    c = ws.cell(row=row, column=1, value="考试统计")
    c.font = TITLE_FONT
    c.alignment = CENTER
    c.fill = TITLE_FILL
    row += 1

    for label, pop in blocks:
        pool = pool_for(pop)
        g1 = stats_of(pool)                       # 参加考试（含往届补考）
        g2 = stats_of([r for r in pool if r["year"] == str(target_year)])  # 仅本年

        # 第二行：左 "参加考试" / 右 "2026级X考试统计"
        ws.cell(row=row, column=1, value="参加考试").font = HEAD_FONT
        ws.cell(row=row, column=1).fill = HEAD_FILL
        ws.cell(row=row, column=5,
                value=f"{target_year}级{label}考试统计").font = HEAD_FONT
        ws.cell(row=row, column=5).fill = HEAD_FILL
        for col in (1, 5):
            ws.cell(row=row, column=col).alignment = CENTER
        row += 1

        # 第三行：表头
        head = ["总人数", "通过", "未通过", "通过率"] * 2
        for j, h in enumerate(head, 1):
            cell = ws.cell(row=row, column=j, value=h)
            cell.font = HEAD_FONT
            cell.fill = HEAD_FILL
            cell.alignment = CENTER
            cell.border = BORDER
        row += 1

        # 第四行：数据
        data = g1[:3] + [g1[3]] + g2[:3] + [g2[3]]
        for j, v in enumerate(data, 1):
            cell = ws.cell(row=row, column=j, value=v)
            cell.alignment = CENTER
            cell.border = BORDER
            if j in (4, 8):
                cell.number_format = "0.00%"
        row += 1
        row += 1  # 空行

    for j, w in enumerate([10, 9, 9, 11, 10, 9, 9, 11], 1):
        ws.column_dimensions[get_column_letter(j)].width = w
    wb.save(wb_path)


def _build_stats(records, folder_name, target_year, pass_line):
    valid = [r for r in records if r["category"] == "valid"]
    retake = [r for r in records if r["category"] == "retake"]
    problem = [r for r in records if r["category"] in ("cuo", "format_error")]
    n_valid = len(valid)
    n_pass = sum(1 for r in valid if r["pts"] >= pass_line)
    return {
        "folder_name": folder_name,
        "target_year": target_year,
        "pass_line": pass_line,
        "总记录数": len(records),
        "本年有效": n_valid,
        "本年有效-通过": n_pass,
        "本年有效-未通过": n_valid - n_pass,
        "本年有效通过率": (n_pass / n_valid) if n_valid else 0,
        "往届补考": len(retake),
        "错考/格式异常": len(problem),
        "学院数": len({r["college"] for r in valid}),
        "错考明细": [(r["name"], r["sid"], r["college"], int(r["pts"]), r["reason"])
                    for r in problem],
    }
